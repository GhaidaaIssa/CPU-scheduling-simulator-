template 
class LinkedList {
private:
    struct Node {
        T data;
        Node* next;
        Node(T val) : data(val), next(nullptr) {}
    };
    Node* head;
    Node* tail;
    int size;

public:
    LinkedList() : head(nullptr), tail(nullptr), size(0) {}
    ~LinkedList() { clear(); }

    void clear() {
        Node* current = head;
        while (current != nullptr) {
            Node* next = current->next;
            delete current;
            current = next;
        }
        head = tail = nullptr;
        size = 0;
    }

    void pushBack(T val) {
        Node* newNode = new Node(val);
        if (!head) {
            head = tail = newNode;
        } else {
            tail->next = newNode;
            tail = newNode;
        }
        size++;
    }

    void pushFront(T val) {
        Node* newNode = new Node(val);
        if (!head) {
            head = tail = newNode;
        } else {
            newNode->next = head;
            head = newNode;
        }
        size++;
    }

    T popFront() {
        if (isEmpty()) throw runtime_error("List is empty");
        Node* temp = head;
        T val = temp->data;
        head = head->next;
        if (!head) tail = nullptr;
        delete temp;
        size--;
        return val;
    }

    T& get(int index) {
        Node* curr = head;
        for (int i = 0; i < index; i++) {
            curr = curr->next;
        }
        return curr->data;
    }

    int getSize() const { return size; }
    bool isEmpty() const { return size == 0; }
};

class PriorityQueue {
private:
    struct PQNode {
        Process proc;
        PQNode* next;
        PQNode(Process p) : proc(p), next(nullptr) {}
    };
    PQNode* head;
    int count;

public:
    PriorityQueue() : head(nullptr), count(0) {}
    ~PriorityQueue() {
        while (head) {
            PQNode* temp = head;
            head = head->next;
            delete temp;
        }
    }

    void enqueue(Process p, int mode) {
        PQNode* newNode = new PQNode(p);
        count++;

        if (!head) {
            head = newNode;
            return;
        }

        bool insertBeforeHead = false;
        if (mode == 1) { // SJF / SRTF
            if (p.remainingTime < head->proc.remainingTime ||
               (p.remainingTime == head->proc.remainingTime && p.arrivalTime < head->proc.arrivalTime)) {
                insertBeforeHead = true;
            }
        } else if (mode == 2) { // Priority
            if (p.priority < head->proc.priority ||
               (p.priority == head->proc.priority && p.arrivalTime < head->proc.arrivalTime)) {
                insertBeforeHead = true;
            }
        }

        if (insertBeforeHead) {
            newNode->next = head;
            head = newNode;
            return;
        }

        PQNode* curr = head;
        while (curr->next) {
            bool insertHere = false;
            if (mode == 1) {
                if (p.remainingTime < curr->next->proc.remainingTime ||
                   (p.remainingTime == curr->next->proc.remainingTime && p.arrivalTime < curr->next->proc.arrivalTime)) {
                    insertHere = true;
                }
            } else if (mode == 2) {
                if (p.priority < curr->next->proc.priority ||
                   (p.priority == curr->next->proc.priority && p.arrivalTime < curr->next->proc.arrivalTime)) {
                    insertHere = true;
                }
            }

            if (insertHere) break;
            curr = curr->next;
        }

        newNode->next = curr->next;
        curr->next = newNode;
    }

    Process dequeue() {
        if (!head) throw runtime_error("Priority Queue is empty");
        PQNode* temp = head;
        Process p = temp->proc;
        head = head->next;
        delete temp;
        count--;
        return p;
    }

    Process& top() { return head->proc; }
    bool isEmpty() const { return head == nullptr; }
    int size() const { return count; }
};

void runFCFS(Process inputProcs[], int n) {
    Process procs[100];
    resetProcesses(inputProcs, procs, n);

    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (procs[j].arrivalTime > procs[j + 1].arrivalTime) {
                swap(procs[j], procs[j + 1]);
            }
        }
    }

    LinkedList gantt;
    int currentTime = 0;

    for (int i = 0; i < n; i++) {
        if (currentTime < procs[i].arrivalTime) {
            currentTime = procs[i].arrivalTime;
        }
        
        GanttBlock block;
        block.processId = procs[i].id;
        block.startTime = currentTime;
        
        currentTime += procs[i].burstTime;
        block.endTime = currentTime;
        gantt.pushBack(block);

        procs[i].completionTime = currentTime;
        procs[i].turnaroundTime = procs[i].completionTime - procs[i].arrivalTime;
        procs[i].waitingTime = procs[i].turnaroundTime - procs[i].burstTime;
    }

    printResults(procs, n, gantt, "First Come First Serve (FCFS)");
}

void runSJFNonPreemptive(Process inputProcs[], int n) {
    Process procs[100];
    resetProcesses(inputProcs, procs, n);

    LinkedList gantt;
    int completed = 0, currentTime = 0;

    while (completed < n) {
        PriorityQueue pq;
        for (int i = 0; i < n; i++) {
            if (procs[i].arrivalTime <= currentTime && !procs[i].isCompleted) {
                pq.enqueue(procs[i], 1);
            }
        }

        if (pq.isEmpty()) {
            currentTime++;
            continue;
        }

        Process currentP = pq.dequeue();
        int idx = -1;
        for (int i = 0; i < n; i++) {
            if (procs[i].id == currentP.id) {
                idx = i;
                break;
            }
        }

        GanttBlock block;
        block.processId = procs[idx].id;
        block.startTime = currentTime;
        currentTime += procs[idx].burstTime;
        block.endTime = currentTime;
        gantt.pushBack(block);

        procs[idx].completionTime = currentTime;
        procs[idx].turnaroundTime = procs[idx].completionTime - procs[idx].arrivalTime;
        procs[idx].waitingTime = procs[idx].turnaroundTime - procs[idx].burstTime;
        procs[idx].isCompleted = true;
        completed++;
    }

    printResults(procs, n, gantt, "Shortest Job First (SJF Non-Preemptive)");
}

void runSJFPreemptive(Process inputProcs[], int n) {
    Process procs[100];
    resetProcesses(inputProcs, procs, n);

    LinkedList gantt;
    int completed = 0, currentTime = 0;
    string prevProcess = "";
    int blockStart = 0;

    while (completed < n) {
        PriorityQueue pq;
        for (int i = 0; i < n; i++) {
            if (procs[i].arrivalTime <= currentTime && procs[i].remainingTime > 0) {
                pq.enqueue(procs[i], 1);
            }
        }

        if (pq.isEmpty()) {
            if (prevProcess != "") {
                gantt.pushBack({prevProcess, blockStart, currentTime});
                prevProcess = "";
            }
            currentTime++;
            continue;
        }

        Process currentP = pq.top();
        int idx = -1;
        for (int i = 0; i < n; i++) {
            if (procs[i].id == currentP.id) {
                idx = i;
                break;
            }
        }

        if (prevProcess != procs[idx].id) {
            if (prevProcess != "") {
                gantt.pushBack({prevProcess, blockStart, currentTime});
            }
            prevProcess = procs[idx].id;
            blockStart = currentTime;
        }

        procs[idx].remainingTime--;
        currentTime++;

        if (procs[idx].remainingTime == 0) {
            procs[idx].completionTime = currentTime;
            procs[idx].turnaroundTime = procs[idx].completionTime - procs[idx].arrivalTime;
            procs[idx].waitingTime = procs[idx].turnaroundTime - procs[idx].burstTime;
            procs[idx].isCompleted = true;
            completed++;
            gantt.pushBack({procs[idx].id, blockStart, currentTime});
            prevProcess = "";
        }
    }

    printResults(procs, n, gantt, "Shortest Remaining Time First (SRTF)");
}

void runRoundRobin(Process inputProcs[], int n, int quantum) {
    Process procs[100];
    resetProcesses(inputProcs, procs, n);

    Queue readyQueue;
    LinkedList gantt;
    bool inQueue[100] = {false};

    int currentTime = 0, completed = 0;

    for (int i = 0; i < n; i++) {
        if (procs[i].arrivalTime == 0) {
            readyQueue.enqueue(i);
            inQueue[i] = true;
        }
    }

    while (completed < n) {
        if (readyQueue.isEmpty()) {
            currentTime++;
            for (int i = 0; i < n; i++) {
                if (procs[i].arrivalTime <= currentTime && !inQueue[i] && procs[i].remainingTime > 0) {
                    readyQueue.enqueue(i);
                    inQueue[i] = true;
                }
            }
            continue;
        }

        int idx = readyQueue.dequeue();
        int execTime = min(quantum, procs[idx].remainingTime);

        GanttBlock block;
        block.processId = procs[idx].id;
        block.startTime = currentTime;
        currentTime += execTime;
        block.endTime = currentTime;
        gantt.pushBack(block);

        procs[idx].remainingTime -= execTime;

        for (int i = 0; i < n; i++) {
            if (i != idx && procs[i].arrivalTime <= currentTime && !inQueue[i] && procs[i].remainingTime > 0) {
                readyQueue.enqueue(i);
                inQueue[i] = true;
            }
        }

        if (procs[idx].remainingTime > 0) {
            readyQueue.enqueue(idx);
        } else {
            procs[idx].completionTime = currentTime;
            procs[idx].turnaroundTime = procs[idx].completionTime - procs[idx].arrivalTime;
            procs[idx].waitingTime = procs[idx].turnaroundTime - procs[idx].burstTime;
            procs[idx].isCompleted = true;
            completed++;
        }
    }

    printResults(procs, n, gantt, "Round Robin (RR, Quantum = " + to_string(quantum) + ")");
}

1. Project Overview & Objectives

This project explores CPU process scheduling in modern operating systems. The core goal is to evaluate process execution performance under different scheduling strategies by implementing custom Data Structures (Queue, Priority Queue, Linked List, and Stack) without using built-in standard template library (STL) scheduling components.

2. Required Data Structures Implementation

Linked List

Serves as the foundation for dynamic process memory allocation without fixed size limits.

Custom Queue (FIFO)

Used in FCFS and Round Robin algorithms to maintain process arrival order.

Priority Queue

Used in SJF, SRTF, and Priority Scheduling to select processes with minimum burst time or highest priority.

Stack Structure

Utilized during preemptive context switches to handle process state recovery.

3. Implemented CPU Scheduling Algorithms

First-Come First-Serve (FCFS): Non-preemptive scheduling where processes are executed strictly in order of arrival using a FIFO Queue.

Shortest Job First (SJF): Non-preemptive algorithm selecting the process with the shortest burst time using a Priority Queue.

Shortest Remaining Time First (SRTF): Preemptive version of SJF where a running process is interrupted if a process with a shorter burst arrives.

Priority Scheduling: Assigns CPU based on priority levels (lower integer = higher priority).

Round Robin (RR): Preemptive algorithm using time slices (quantum) to ensure fair CPU allocation across processes.

5. Performance Analysis & Conclusion

Based on the simulation results, preemptive scheduling algorithms (such as SRTF and Priority Preemptive) generally reduce average waiting time by preventing long jobs from blocking shorter high-priority processes (Convoy Effect). Round Robin provides optimal interactive response times, making it suitable for multi-user systems.
